Written by sstack setup during onboarding. factory-verify runs the verify list at the pushed commit. Runtimes, install and secret names stay in .factory/setup.json.

# .sstack/project.toml
brief  = ".sstack/product.md"
verify = ["pnpm lint", "pnpm typecheck", "pnpm test"]
graph  = true
check  = "pnpm eslint {path}"

[test.code]
framework = "jest, *.test.ts next to the module"
run       = "pnpm jest {path}"

[test.mobile]
framework = "maestro, .maestro/*.yaml"
build     = "npx expo run:ios"
run       = "maestro test {path}"

[practices]
"src/screens/*" = ["react-native-way"]
